// Creating a duplicate table from an existing table:

# this considers both the structure and data of the existing table and creates a new table with the same structure and data.
syntax : create table new_table_name as select * from existing_table_name;
example : create table backup_employees as select * from employees;

# if only the structure of the existing table is to be copied, then the following syntax can be used:
syntax : create table new_table_name like existing_table_name;
or
syntax : create table new_table_name as select * from existing_table_name where 1=0;

example : create table backup_employees as select * from employees where 1=0;
